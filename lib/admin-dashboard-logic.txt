function log(elId, msg, type = '') {
  const el = document.getElementById(elId);
  if (!el) return;
  el.style.display = 'block';
  const cls = { ok: 'log-ok', err: 'log-err', info: 'log-info', warn: 'log-warn' }[type] || '';
  el.innerHTML += `<div class="${cls}">${new Date().toLocaleTimeString('ko-KR')} ${escHtml(msg)}</div>`;
  el.scrollTop = el.scrollHeight;
}

async function callApi(path, body, logId, authMode = 'ADMIN') {
  const token = await getAuthToken();
  if (!token) { showToast('로그인이 필요합니다', 'error'); return null; }

  try {
    const res = await fetch(path, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${token}` },
      body: body && Object.keys(body).length ? JSON.stringify(body) : undefined,
    });
    const data = await res.json();
    if (!res.ok) { log(logId, `오류: ${data.error || res.status}`, 'err'); return null; }
    return data;
  } catch(e) { log(logId, `네트워크 오류: ${e.message}`, 'err'); return null; }
}

async function runAnalyze(issueId = null, limit = 3, logId = 'analyzeLog', maxChain = null) {
  const el = document.getElementById(logId);
  if (el) el.innerHTML = '';
  const totalLabel = maxChain ? `최대 ${maxChain * 5}건 (체이닝, 백그라운드에서 계속 진행)` : `최대 ${limit}건`;
  log(logId, `분석 시작 (${totalLabel})...`, 'info');

  const body = issueId ? { issue_id: issueId } : { limit, ...(maxChain ? { max_chain: maxChain } : {}) };
  const data = await callApi('/api/analyze', body, logId);
  if (!data) return;

  if (maxChain) {
    log(logId, `1차 배치 완료: ${data.analyzed}건 — 서버가 최대 ${(data.maxChain || maxChain) * 5}건까지 자동으로 이어서 처리 중입니다 (몇 분 후 이슈 목록에서 진행 확인)`, 'ok');
  } else {
    log(logId, `분석 완료: ${data.analyzed}건`, 'ok');
  }
  if (data.errors?.length) data.errors.forEach(e => log(logId, e, 'warn'));
  showToast(maxChain ? `1차 배치 ${data.analyzed}건 완료 — 백그라운드에서 계속 진행 중` : `${data.analyzed}건 분석 완료`, 'success');
}

async function loadCompaniesTable(autoFetch = true) {
  const tbody = document.getElementById('companiesTable');
  const { data, error } = await sb.from('companies').select('*').order('created_at', { ascending: false }).limit(100);
  if (error || !data?.length) { tbody.innerHTML = `<tr><td colspan="7" class="loading-row">${error?.message || '데이터 없음'}</td></tr>`; return; }

  tbody.innerHTML = data.map(c => {
    const fmtPrice = c.current_price ? `${c.currency === 'KRW' ? '₩' : '$'}${Number(c.current_price).toLocaleString()}` : '—';
    const fmtMc = c.market_cap ? (c.market_cap >= 1e12 ? `$${(c.market_cap/1e12).toFixed(1)}T` : c.market_cap >= 1e9 ? `$${(c.market_cap/1e9).toFixed(1)}B` : `$${(c.market_cap/1e6).toFixed(0)}M`) : '—';
    const updated = c.price_updated_at ? new Date(c.price_updated_at).toLocaleDateString('ko-KR') : '—';
    return `<tr id="tr-${escHtml(c.ticker)}">
      <td class="ticker-cell">${escHtml(c.ticker)}</td>
      <td>${escHtml(c.name_ko || c.name_en || '—')}</td>
      <td><span class="badge ${c.market === 'KR' ? 'badge-blue' : 'badge-gray'}">${c.market || '—'}</span></td>
      <td id="price-${escHtml(c.ticker)}" style="color:var(--text2)">${fmtPrice}</td>
      <td style="color:var(--text2)">${fmtMc}</td>
      <td style="font-size:11px;color:var(--text3)">${updated}</td>
      <td><div class="row-actions">
        <button class="icon-btn" onclick="refreshPrice('${escHtml(c.ticker)}')" title="주가 갱신">
          <svg width="13" height="13" fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><polyline points="1 4 1 10 7 10"/><path d="M3.51 15a9 9 0 1 0 .49-4.61"/></svg>
        </button>
      </div></td>
    </tr>`;
  }).join('');

  // 가격 없는 기업 자동 조회 (순차, 0.5초 간격)
  if (autoFetch) {
    const noPrice = data.filter(c => !c.current_price);
    for (const c of noPrice) {
      await refreshPrice(c.ticker, true);
      await new Promise(r => setTimeout(r, 500));
    }
  }
}

async function refreshAllPrices() {
  const { data } = await sb.from('companies').select('ticker').limit(100);
  if (!data?.length) return;
  showToast(`${data.length}개 주가 갱신 시작...`, 'info');
  for (const c of data) {
    await refreshPrice(c.ticker, true);
    await new Promise(r => setTimeout(r, 400));
  }
  showToast('전체 갱신 완료', 'success');
  loadCompaniesTable(false);
}

async function refreshPrice(ticker, silent = false) {
  if (!silent) showToast(`${ticker} 갱신 중...`, 'info');
  try {
    const res = await fetch(`/api/stock-price?ticker=${encodeURIComponent(ticker)}`);
    const data = await res.json();
    if (data.price) {
      const cell = document.getElementById(`price-${ticker}`);
      if (cell) cell.textContent = `${data.currency === 'KRW' ? '₩' : '$'}${Number(data.price).toLocaleString()}`;
      if (!silent) showToast(`${ticker}: ${data.price}`, 'success');
    }
  } catch(e) { if (!silent) showToast(e.message, 'error'); }
}

function showToast(msg, type = 'info') {
  const t = document.getElementById('toast');
  const icons = { success: '✓', error: '✗', info: 'ℹ', warn: '⚠' };
  t.className = `toast ${type}`;
  t.innerHTML = `<span>${icons[type] || ''}</span><span>${escHtml(msg)}</span>`;
  t.classList.add('show');
  setTimeout(() => t.classList.remove('show'), 3000);
}

function escHtml(s) {
  return String(s || '').replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;');
}

// 2-패스 애널리스트 파이프라인 배포 시각 — 구형/신형 구간 분리 기준
const PIPELINE_V2_AT = '2026-07-02T13:43';

async function loadAccuracy() {
  const periods = ['1d', '3d', '7d', '30d'];

  for (const p of periods) {
    // count 쿼리 사용: select-rows 방식은 PostgREST 기본 1000행 캡에 걸려 총계가 잘림
    const cnt = (mods) => {
      let q = sb.from('analysis_companies').select('id', { count: 'exact', head: true }).not(`check_date_${p}`, 'is', null);
      return mods ? mods(q) : q;
    };
    const [total, hit, nTotal, nHit] = await Promise.all([
      cnt(),
      cnt(q => q.eq(`is_accurate_${p}`, true)),
      cnt(q => q.gte('entry_date', PIPELINE_V2_AT)),
      cnt(q => q.eq(`is_accurate_${p}`, true).gte('entry_date', PIPELINE_V2_AT)),
    ]);
    document.getElementById(`acc-${p}-total`).textContent = (total.count ?? 0).toLocaleString();
    document.getElementById(`acc-${p}-rate`).textContent = total.count
      ? Math.round((hit.count || 0) / total.count * 100) + '%' : '—';

    const splitEl = document.getElementById(`acc-${p}-split`);
    if (splitEl) {
      if (nTotal.count) {
        const oldT = (total.count || 0) - nTotal.count;
        const oldH = (hit.count || 0) - (nHit.count || 0);
        const newR = Math.round((nHit.count || 0) / nTotal.count * 100);
        splitEl.textContent = `구형 ${oldT ? Math.round(oldH / oldT * 100) + '%' : '—'} (${oldT.toLocaleString()}) → 신형 ${newR}% (${nTotal.count.toLocaleString()})`;
      } else {
        splitEl.textContent = '신형 파이프라인 검증 대기';
      }
    }

    const { data: rows } = await sb.from('analysis_companies')
      .select(`upside_pct, entry_price, check_price_${p}, actual_return_${p}, is_accurate_${p}, companies(ticker, name_ko, currency)`)
      .not(`check_date_${p}`, 'is', null)
      .order('entry_date', { ascending: false }).limit(50);
    renderAccTable(`accuracy${p}Table`, rows, p);
  }
}

function renderAccTable(tbodyId, rows, period) {
  const tbody = document.getElementById(tbodyId);
  const labelMap = { '1d': '1일', '3d': '3일', '7d': '7일', '30d': '30일' };
  if (!rows?.length) { tbody.innerHTML = `<tr><td colspan="7" class="loading-row">아직 데이터 없음 (분석 후 ${labelMap[period]} 경과 시 자동 집계)</td></tr>`; return; }
  const sym = r => r.companies?.currency === 'KRW' ? '₩' : '$';
  tbody.innerHTML = rows.map(r => {
    const ret = r[`actual_return_${period}`];
    const retColor = ret > 0 ? 'var(--green)' : ret < 0 ? 'var(--red)' : 'var(--text2)';
    const acc = r[`is_accurate_${period}`];
    const accBadge = acc == null ? '<span class="badge badge-gray">—</span>'
      : acc ? '<span class="badge badge-green">✓ 적중</span>'
      : '<span class="badge badge-red">✗ 빗나감</span>';
    return `<tr>
      <td class="ticker-cell">${escHtml(r.companies?.ticker || '—')}</td>
      <td>${escHtml(r.companies?.name_ko || '—')}</td>
      <td style="color:var(--green)">+${r.upside_pct ?? '—'}%</td>
      <td style="color:var(--text2)">${r.entry_price ? sym(r) + Number(r.entry_price).toLocaleString() : '—'}</td>
      <td style="color:var(--text2)">${r[`check_price_${period}`] ? sym(r) + Number(r[`check_price_${period}`]).toLocaleString() : '—'}</td>
      <td style="color:${retColor};font-weight:600">${ret != null ? (ret > 0 ? '+' : '') + ret + '%' : '—'}</td>
      <td>${accBadge}</td>
    </tr>`;
  }).join('');
}

async function reanRunSingle() {
  const btn = document.getElementById('reanRunBtn');
  const count = parseInt(document.getElementById('reanCount').value, 10) || 5;
  btn.disabled = true;
  const orig = btn.textContent;
  btn.textContent = '실행 중...';
  reanLog(`🔄 최근 ${count}개 재분석 시작 (구버전 삭제 + 새 형식 분석)`, 'info');
  startLiveCounter(`재분석 진행 중 (~${count * 8}초 예상)`);
  const result = await callApi('/api/analyze', { force_recent: count, limit: count }, 'reanLog');
  stopLiveCounter();
  if (result) {
    reanLog(`✓ 완료: 삭제 ${result.reanalyzed || 0}건, 새 분석 ${result.analyzed || 0}건`, 'ok');
    for (const p of (result.processed || [])) {
      const dur = Math.round((p.duration_ms || 0) / 100) / 10;
      reanLog(`  ✓ ${p.title} (${dur}s) → ${(p.companies || []).join(', ')}`, 'ok');
    }
    if (result.errors?.length) result.errors.slice(0, 5).forEach(e => reanLog(`  ⚠️ ${e}`, 'warn'));
  }
  btn.disabled = false;
  btn.textContent = orig;
}

let _reanAutoStop = false;
let _reanLiveTimer = null;
function startLiveCounter(label) {
  const start = Date.now();
  stopLiveCounter();
  const live = document.createElement('div');
  live.id = 'reanLiveLine';
  live.style.cssText = 'color:var(--blue);font-weight:600;padding:4px 0';
  document.getElementById('reanLog').appendChild(live);
  const update = () => {
    const sec = Math.round((Date.now() - start) / 1000);
    const dots = '.'.repeat((sec % 4));
    live.textContent = `⏳ ${label} ${sec}초 경과${dots}`;
  };
  update();
  _reanLiveTimer = setInterval(update, 250);
  return start;
}
function stopLiveCounter() {
  if (_reanLiveTimer) { clearInterval(_reanLiveTimer); _reanLiveTimer = null; }
  document.getElementById('reanLiveLine')?.remove();
}

// 회사명 보정
function fnTag(msg, color = 'var(--text2)') {
  const el = document.getElementById('fixNamesLog');
  el.style.display = 'block';
  const div = document.createElement('div');
  div.style.color = color;
  div.textContent = `${new Date().toLocaleTimeString('ko-KR')} ${msg}`;
  el.appendChild(div);
  el.scrollTop = el.scrollHeight;
}
function fnRenderResult(result, dryRun) {
  fnTag(`✓ 스캔 ${result.scanned}개 중 ${result.updated}개 변경 ${dryRun ? '예정' : '완료'}`, 'var(--green)');
  (result.updates || []).forEach(u => {
    const before = `[${u.before.ko || '∅'} / ${u.before.en || '∅'}]`;
    const after  = `[${u.after.name_ko || u.before.ko || '∅'} / ${u.after.name_en || u.before.en || '∅'}]`;
    fnTag(`  ${u.ticker}: ${before} → ${after}  ${u.reason ? '(' + u.reason + ')' : ''}`);
  });
  (result.errors || []).slice(0, 5).forEach(e => fnTag(`⚠️ ${e}`, 'var(--red)'));
}

// 실시간 상태 박스 제어
let _fnElapsedTimer = null;
let _fnStop = false;
function fnShowStatus(text) {
  document.getElementById('fnStatus').style.display = 'block';
  document.getElementById('fnStatusText').textContent = text;
}
function fnHideStatus() {
  document.getElementById('fnStatus').style.display = 'none';
  if (_fnElapsedTimer) clearInterval(_fnElapsedTimer);
  _fnElapsedTimer = null;
}
function fnStartElapsed() {
  const start = Date.now();
  if (_fnElapsedTimer) clearInterval(_fnElapsedTimer);
  document.getElementById('fnElapsed').textContent = '0';
  _fnElapsedTimer = setInterval(() => {
    document.getElementById('fnElapsed').textContent = Math.round((Date.now() - start) / 1000);
  }, 250);
}
function fnSetProgress(curr, total) {
  document.getElementById('fnBatchN').textContent = curr;
  document.getElementById('fnBatchT').textContent = total;
  document.getElementById('fnProgBar').style.width = total ? `${Math.min(100, (curr/total)*100)}%` : '0';
}

// AI 검증 모드: 작은 배치 자동 반복 + 실시간 상태
async function fixNamesAi(dryRun) {
  const log = document.getElementById('fixNamesLog');
  log.innerHTML = '';
  _fnStop = false;
  const previewBtn = document.getElementById('fnAiPreviewBtn');
  const runBtn     = document.getElementById('fnAiRunBtn');
  const targetBtn  = dryRun ? previewBtn : runBtn;
  const origText = targetBtn.textContent;
  targetBtn.textContent = '🛑 중단';
  targetBtn.onclick = () => { _fnStop = true; fnTag('🛑 중단 요청, 현재 배치 후 종료', 'var(--yellow)'); };
  previewBtn.disabled = dryRun ? false : true;
  runBtn.disabled = dryRun ? true : false;

  const batchSize = Math.max(3, Math.min(20, parseInt(document.getElementById('fnBatchSize').value, 10) || 10));
  fnTag(dryRun ? `🔍 AI 검증 미리보기 (배치 1, ${batchSize}개)` : `🤖 AI 검증 자동 반복 시작 (배치 ${batchSize}개씩)`, 'var(--blue)');
  fnShowStatus('첫 배치 시작...');
  fnStartElapsed();

  let offset = 0;
  let totalUpdated = 0;
  let totalProcessed = 0;
  let totalKnown = null;
  let batch = 1;

  while (!_fnStop) {
    fnShowStatus(`📦 배치 ${batch} 처리 중 (offset ${offset}, ${batchSize}개)...`);
    fnSetProgress(batch - 1, totalKnown ? Math.ceil(totalKnown / batchSize) : '?');
    fnTag(`📦 배치 ${batch} (offset ${offset}, ${batchSize}개)`, 'var(--text3)');

    const result = await callApi('/api/fix-company-names',
      { dry_run: dryRun, ai_verify: true, batch_size: batchSize, batch_offset: offset },
      'fixNamesLog');
    if (!result) { fnTag(`❌ 배치 ${batch} 실패 (네트워크 또는 타임아웃) — 배치 크기 줄여 재시도하세요`, 'var(--red)'); break; }

    if (totalKnown == null) totalKnown = result.total;
    totalProcessed += result.scanned || 0;
    totalUpdated   += result.updated || 0;

    fnRenderResult(result, dryRun);
    if (result.timing) {
      fnTag(`  ⏱ Yahoo ${(result.timing.yhMs/1000).toFixed(1)}s · Claude ${(result.timing.aiMs/1000).toFixed(1)}s · 총 ${(result.timing.totalMs/1000).toFixed(1)}s`, 'var(--text3)');
    }
    fnSetProgress(batch, Math.ceil(totalKnown / batchSize));

    if (dryRun) break;
    if (result.nextOffset == null) {
      fnTag(`🎯 완료: 전체 ${totalUpdated}개 변경 (스캔 ${totalProcessed}/${totalKnown})`, 'var(--green)');
      break;
    }
    offset = result.nextOffset;
    batch++;
    await new Promise(r => setTimeout(r, 800));
  }

  if (_fnStop) fnTag(`🛑 중단됨: ${totalUpdated}개 변경됨, 다음 offset ${offset}`, 'var(--yellow)');
  fnHideStatus();
  targetBtn.textContent = origText;
  targetBtn.onclick = () => (dryRun ? fixNamesAi(true) : fixNamesAi(false));
  previewBtn.disabled = false;
  runBtn.disabled = false;
}

async function loadBacktest() {
  const token = await getAuthToken();
  if (!token) { showToast('로그인 필요', 'error'); return; }
  showToast('통계 집계 중...', 'info');
  try {
    const res = await fetch('/api/stats-backtest', {
      method: 'GET',
      headers: { 'Authorization': `Bearer ${token}` },
    });
    if (!res.ok) { showToast(`HTTP ${res.status}`, 'error'); return; }
    const d = await res.json();
    if (d.empty) {
      document.getElementById('btUpdated').textContent = '검증된 데이터 없음 — 분석 + 시간 경과 필요';
      return;

    document.getElementById('btUpdated').textContent = '업데이트: ' + new Date().toLocaleTimeString('ko-KR');
  } catch(e) {
    showToast(`에러: ${e.message}`, 'error');
  }
}

function accColor(v) {
  if (v == null) return 'var(--text3)';
  if (v >= 70) return 'var(--green)';
  if (v >= 50) return 'var(--yellow)';
  return 'var(--red)';
}
function retColor(v) {
  if (v == null) return 'var(--text3)';
  if (v > 0) return 'var(--green)';
  if (v < 0) return 'var(--red)';
  return 'var(--text2)';
}
function fmtPctVal(v) {
  if (v == null || isNaN(v)) return '—';
  const rounded = Math.round(v * 100) / 100;  // floating point 정리
  return (rounded > 0 ? '+' : '') + rounded.toFixed(2) + '%';
}

async function invLoad(page = 1) {
  _invPage = page;
  const status = document.getElementById('invStatusFilter').value;
  const q = document.getElementById('invSearch').value.trim();
  const params = new URLSearchParams({ action: 'list-investments', status, page, page_size: _invPageSize });
  if (q) params.set('q', q);

  const box = document.getElementById('invList');
  box.innerHTML = '<div style="color:var(--text3);font-size:13px;padding:20px;text-align:center">불러오는 중...</div>';

  try {
    const r = await fetch('/api/admin?' + params.toString(), {
      headers: { Authorization: 'Bearer ' + (await getAuthToken()) },
    });
    const j = await r.json();
    if (!j.ok) throw new Error(j.error || 'list failed');

    // 티커 셀렉트 옵션 업데이트
    const sel = document.getElementById('invChartTicker');
    const curVal = sel.value;
    sel.innerHTML = '<option value="">티커 선택...</option>' +

    // 목록 렌더
    if (!j.items.length) {
      box.innerHTML = '<div style="color:var(--text3);font-size:13px;padding:30px;text-align:center;background:var(--bg2);border-radius:10px">아직 추출된 항목이 없습니다. "즉시 추출" 또는 "DART 폴링" 버튼을 눌러 시작하세요.</div>';
    } else {
      box.innerHTML = j.items.map(invRowHtml).join('');
    }

    // 페이지네이션
    const totalPages = Math.ceil(j.total / _invPageSize);
    const pg = document.getElementById('invPagination');
    if (totalPages <= 1) { pg.innerHTML = ''; }
    else {
      let html = `<button class="btn btn-ghost" ${_invPage===1?'disabled':''} onclick="invLoad(${_invPage-1})">← 이전</button>`;
      html += `<span style="padding:6px 12px;color:var(--text2);font-size:13px">${_invPage} / ${totalPages} · 총 ${j.total}건</span>`;
      html += `<button class="btn btn-ghost" ${_invPage===totalPages?'disabled':''} onclick="invLoad(${_invPage+1})">다음 →</button>`;
      pg.innerHTML = html;
    }
  } catch (e) {
    box.innerHTML = `<div style="color:var(--red);padding:20px">로드 실패: ${e.message}</div>`;
  }
}

async function invLoadChart() {
  const ticker = document.getElementById('invChartTicker').value;
  const hint = document.getElementById('invChartHint');
  if (!ticker) { hint.textContent = '상단의 티커를 선택하면 추출 이벤트 누적 추이가 표시됩니다.'; return; }
  hint.textContent = '로딩 중...';

  try {
    const r = await fetch('/api/admin?action=investment-events&ticker=' + encodeURIComponent(ticker));
    const j = await r.json();
    if (!j.ok) throw new Error(j.error || 'fetch failed');

    const series = j.series || [];
    if (!series.length) {
      hint.textContent = '이 티커의 이벤트가 아직 없습니다.';
      if (_invChart) { _invChart.destroy(); _invChart = null; }
      return;
    }

    const palette = ['#0066cc','#e31937','#00873a','#7c3aed','#b8860b','#06b6d4','#ec4899','#f97316','#84cc16','#a855f7'];
    const datasets = series.slice(0, 10).map((s, i) => ({
      label: s.target,
      data: s.points.map(p => ({ x: p.t, y: p.cumulative })),
      borderColor: palette[i % palette.length],
      backgroundColor: palette[i % palette.length] + '22',
      borderWidth: 2, pointRadius: 3, tension: 0.2, stepped: 'before',
    }));

    const ctx = document.getElementById('invChart');
    if (_invChart) _invChart.destroy();
    _invChart = new Chart(ctx, {
      type: 'line',
      data: { datasets },
      options: {
        responsive: true, maintainAspectRatio: false,
        plugins: {
          legend: { position: 'bottom', labels: { color: '#4b4b52', boxWidth: 12, font: { size: 11 } } },
          tooltip: {
            backgroundColor: '#fff', borderColor: '#e5e5ea', borderWidth: 1,
            titleColor: '#0a0a0a', bodyColor: '#4b4b52',
            callbacks: {
              title: (items) => new Date(items[0].parsed.x).toLocaleString('ko-KR'),
              label: (item) => `${item.dataset.label}: ${item.parsed.y}회`,
            },
          },
        },
        scales: {
          x: {
            type: 'time',
            time: { unit: 'day', displayFormats: { day: 'MM/dd' } },
            ticks: { color: '#86868b', font: { size: 10 } },
            grid: { color: '#e5e5ea' },
          },
          y: {
            beginAtZero: true,
            ticks: { color: '#86868b', font: { size: 10 }, stepSize: 1 },
            grid: { color: '#e5e5ea' },
            title: { display: true, text: '누적 추출 횟수', color: '#4b4b52', font: { size: 11 } },
          },
        },
      },
    });
    hint.textContent = `${ticker}: ${series.length}개 대상, 총 ${j.events.length}개 이벤트`;
  } catch (e) {
    hint.textContent = '❌ ' + e.message;
  }
}

async function loadFeedback() {
  const status = document.getElementById('fbStatusFilter')?.value || 'new';
  const list = document.getElementById('fbList');
  if (!list) return;
  list.innerHTML = '<div class="loading-row">불러오는 중...</div>';
  const data = await callApi(`/api/feedback?action=list&status=${status}`, {}, 'fbLog');
  if (!data) { list.innerHTML = '<div class="loading-row">불러오기 실패</div>'; return; }
  const items = data.items || [];
  const countEl = document.getElementById('fbCount');
  if (countEl) countEl.textContent = `${items.length}건`;
  if (!items.length) { list.innerHTML = '<div class="loading-row">해당 상태의 피드백이 없습니다</div>'; return; }
  list.innerHTML = items.map(renderFeedbackCard).join('');
}

      <span class="fb-cat-badge" style="background:${FB_CAT_COLOR[f.category] || 'var(--bg3)'}">${FB_CAT_LABEL[f.category] || escHtml(f.category)}</span>

function renderUsersFiltered() {
  const q = (document.getElementById('usersSearch')?.value || '').trim().toLowerCase();
  const tbody = document.getElementById('usersTableBody');
  if (!tbody) return;
  const filtered = q ? _usersCache.filter(u => (u.email || '').toLowerCase().includes(q)) : _usersCache;

  if (!filtered.length) {
    tbody.innerHTML = '<tr><td colspan="6" style="text-align:center;color:var(--text3)">데이터 없음</td></tr>';
    return;
  }

  tbody.innerHTML = filtered.map(u => `
    <tr>
      <td>${escHtml(u.email || '-')}</td>
      <td>${u.provider === 'google' ? 'Google' : '이메일'}</td>
      <td style="font-size:12px;color:var(--text3)">${u.created_at ? new Date(u.created_at).toLocaleDateString('ko-KR') : '-'}</td>
      <td style="font-size:12px;color:var(--text3)">${u.last_sign_in_at ? new Date(u.last_sign_in_at).toLocaleDateString('ko-KR') : '-'}</td>

        <button class="btn-sm ${u.banned ? 'btn-ghost' : 'btn-danger'}" onclick="toggleBanUser('${u.id}', ${!u.banned})">${u.banned ? '차단 해제' : '차단'}</button>

  `).join('');
}

async function toggleBanUser(userId, ban) {
  if (ban && !confirm('이 회원을 차단하시겠습니까? 로그인이 즉시 막힙니다.')) return;
  const data = await callApi('/api/admin?action=ban-user', { user_id: userId, ban }, null);

  showToast(ban ? '차단했습니다' : '차단을 해제했습니다', 'success');
  loadUsers();
}

// ── 실시간 접속자 현황 (Supabase Realtime Presence) ───────────────
// 공개 페이지(index.html/company.html/analysis.html)가 'site-presence' 채널에
// track()으로 자신을 등록하고, 여기서는 track 없이 구독만 해서 그 상태를 관찰한다
// (관리자 본인의 admin 페이지 접속은 방문자 수에 포함되지 않음).
let _presenceCh = null;
function initPresenceWatch() {
  if (_presenceCh) return;
  _presenceCh = sb.channel('site-presence');
  _presenceCh.on('presence', { event: 'sync' }, () => {
    const state = _presenceCh.presenceState();
    const keys = Object.keys(state);
    const el = document.getElementById('ds-live-visitors');
    if (el) el.textContent = keys.length;
    const pages = {};
    for (const k of keys) {
      const p = state[k]?.[0]?.page || '?';
      pages[p] = (pages[p] || 0) + 1;
    }
    const pagesEl = document.getElementById('ds-live-pages');
    if (pagesEl) {
      const sorted = Object.entries(pages).sort((a, b) => b[1] - a[1]);
      pagesEl.innerHTML = sorted.length
        ? sorted.map(([p, n]) => `${escHtml(p)} · ${n}명`).join('<br>')
        : '접속자 없음';
    }
  });
  _presenceCh.subscribe();
}
